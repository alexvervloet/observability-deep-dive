# The default stack (PROVIDER=mock) needs only python-dotenv: everything else
# is the Python standard library. That's deliberate: the whole observability
# stack runs offline, with no key and almost no dependencies. The traffic
# simulator, every metric, every drift detector, the alerting, and the capstone
# dashboard all run on synthetic logs the repo generates itself.

# Loads variables from a .env file (like PROVIDER) into the environment.
python-dotenv>=1.0.0

# --- Provider SDKs (OPTIONAL) ---
# You only need these if you switch PROVIDER away from the offline mock: e.g. to
# score a sample of answers with a *real* LLM-as-judge (the quality-drift section)
# or to compute *real* embeddings for input-drift. Both have deterministic offline
# mocks, so nothing here is required to complete the repo.

# OpenAI: used when PROVIDER=openai.
openai>=2.0,<3

# Anthropic: used when PROVIDER=claude.
anthropic>=0.111.0,<1

# --- OpenTelemetry (OPTIONAL: Section 11 only) ---
# The one section that emits real telemetry instead of analyzing logs. The SDK
# runs fully offline (its console and in-memory exporters need no network); the
# OTLP exporter is what puts spans on the wire, to hands_on/otel_collector.py, a
# real collector, or a vendor. Nothing else in the repo imports these.
opentelemetry-sdk>=1.44,<2
opentelemetry-exporter-otlp-proto-http>=1.44,<2
